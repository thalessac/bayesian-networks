\begin{frame}[t]{Algoritmo PC: fundamentos}
  \framesubtitle{Peter Spirtes y Clark Glymour (1991): dos fases a partir de los datos.}
  \begin{center}
    \begin{tikzpicture}[node distance=0.55cm]
      \node[flow,text width=2.6cm] (datos) {\textbf{Datos}\\[2pt]
        {\footnotesize Observaciones de las variables}};
      \node[flow,text width=2.6cm,right=of datos] (f1) {\textbf{Fase 1}\\[2pt]
        {\footnotesize Esqueleto: qué está conectado}};
      \node[flow,text width=2.6cm,right=of f1] (f2) {\textbf{Fase 2}\\[2pt]
        {\footnotesize Orientación: hacia dónde van las flechas}};
      \node[flow,text width=2.6cm,right=of f2,fill=upmorange!10,draw=upmorange]
        (cpdag) {\textbf{CPDAG}\\[2pt]
        {\footnotesize Grafo parcialmente dirigido}};
      \draw[flow arrow] (datos) -- (f1);
      \draw[flow arrow] (f1) -- (f2);
      \draw[flow arrow] (f2) -- (cpdag);
    \end{tikzpicture}
  \end{center}
  \begin{itemize}
    \item Supuestos: condición de Markov causal, fidelidad y suficiencia
          causal (sin variables ocultas).
    \item Fase 1: parte del grafo completo y poda aristas con contrastes de
          independencia, guardando el conjunto separador de cada par.
    \item Fase 2: orienta $v$-estructuras (colisores) y propaga con las reglas
          de Meek.
  \end{itemize}
  \begin{alertblock}{La idea que lo hace eficiente}
    Solo condiciona sobre los vecinos actuales de cada variable, no sobre
    todos los subconjuntos posibles (como hacía SGS). Coste exponencial en el
    peor caso, pero polinómico si el grafo es poco denso.
  \end{alertblock}
\end{frame}

\begin{frame}[t]{Fase 1: el esqueleto}
  \framesubtitle{Todo conectado al inicio; se eliminan aristas con pruebas.}
  {\small\color{upmgray}Grafo verdadero (desconocido): $A\to C\leftarrow B$,
  $C\to D$.}
  \vspace{0.2cm}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.31\textwidth}
      \centering\textbf{\color{upmblue}1. Grafo completo}\\[0.25cm]
      \begin{tikzpicture}[x=1.1cm,y=0.9cm]
        \node[bn small] (A) at (0,2) {$A$};
        \node[bn small] (B) at (2,2) {$B$};
        \node[bn small] (C) at (1,1) {$C$};
        \node[bn small] (D) at (1,0) {$D$};
        \draw[skel] (A) -- (B);
        \draw[skel] (A) -- (C);
        \draw[skel] (B) -- (C);
        \draw[skel] (C) -- (D);
        \draw[skel] (A) to[bend right=25] (D);
        \draw[skel] (B) to[bend left=25] (D);
      \end{tikzpicture}\\[0.25cm]
      {\small Todo conectado.}
    \end{column}
    \begin{column}{0.31\textwidth}
      \centering\textbf{\color{upmblue}2. Orden $\ell=0$}\\[0.25cm]
      \begin{tikzpicture}[x=1.1cm,y=0.9cm]
        \node[bn small] (A) at (0,2) {$A$};
        \node[bn small] (B) at (2,2) {$B$};
        \node[bn small] (C) at (1,1) {$C$};
        \node[bn small] (D) at (1,0) {$D$};
        \draw[removed] (A) -- (B);

        \draw[skel] (C) -- (D);

      \end{tikzpicture}\\[0.25cm]
      {\small $A\perp B$ $\Rightarrow$ se elimina $A$--$B$\\
      $S(A,B)=\emptyset$}
    \end{column}
    \begin{column}{0.31\textwidth}
      \centering\textbf{\color{upmblue}3. Orden $\ell=1$}\\[0.25cm]
      \begin{tikzpicture}[x=1.1cm,y=0.9cm]
        \node[bn small] (A) at (0,2) {$A$};
        \node[bn small] (B) at (2,2) {$B$};
        \node[bn small] (C) at (1,1) {$C$};
        \node[bn small] (D) at (1,0) {$D$};
        \draw[skel] (A) -- (C);
        \draw[skel] (B) -- (C);
        \draw[skel] (C) -- (D);
        \draw[removed] (A) to[bend right=25] (D);
        \draw[removed] (B) to[bend left=25] (D);
      \end{tikzpicture}\\[0.25cm]

    \end{column}
  \end{columns}
  \vfill
  {\scriptsize\color{upmgray}Líneas discontinuas: aristas eliminadas en cada
  paso. Se guarda el conjunto separador de cada par, porque la fase 2 lo
  usa.}
\end{frame}

\begin{frame}[t]{Fase 2: orientar las flechas}
  \framesubtitle{Primero los colisores, después reglas de propagación.}
  \vspace{0.3cm}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.30\textwidth}
      \centering\textbf{\color{upmblue}1. Colisor}\\[0.25cm]
      \begin{tikzpicture}[x=1.1cm,y=0.9cm]
        \node[bn small] (A) at (0,2) {$A$};
        \node[bn small] (B) at (2,2) {$B$};
        \node[bn small] (C) at (1,1) {$C$};
        \node[bn small] (D) at (1,0) {$D$};
        \draw[bn edge,upmorange] (A) -- (C);
        \draw[bn edge,upmorange] (B) -- (C);
        \draw[skel] (C) -- (D);
      \end{tikzpicture}\\[0.25cm]
      {\small $C\notin S(A,B)=\emptyset$\\
      $\Rightarrow A\to C\leftarrow B$}
    \end{column}
    \begin{column}{0.30\textwidth}
      \centering\textbf{\color{upmblue}2. Propagación (Meek)}\\[0.25cm]
      \begin{tikzpicture}[x=1.1cm,y=0.9cm]
        \node[bn small] (A) at (0,2) {$A$};
        \node[bn small] (B) at (2,2) {$B$};
        \node[bn small] (C) at (1,1) {$C$};
        \node[bn small] (D) at (1,0) {$D$};
        \draw[bn edge] (A) -- (C);
        \draw[bn edge] (B) -- (C);
        \draw[bn edge,upmorange] (C) -- (D);
      \end{tikzpicture}\\[0.25cm]
      {\small Sin ciclos ni colisores nuevos\\
      $\Rightarrow C\to D$}
    \end{column}
    \begin{column}{0.34\textwidth}
      \begin{block}{Resultado: un CPDAG}
        Las aristas que los datos no permiten orientar quedan sin dirección:
        \[
          X\to Y\to Z \quad\text{y}\quad X\leftarrow Y\leftarrow Z
        \]
        implican las mismas independencias.
      \end{block}
    \end{column}
  \end{columns}
  \vfill
  {\scriptsize\color{upmgray}Intuición: si $C$ fuera intermediario o causa
  común, habría hecho falta $C$ para separar $A$ y $B$.}
\end{frame}

% ---------------------------------------------------------------------------
\begin{frame}[t]{(i) Alta dimensión}
  \framesubtitle{¿Funciona el PC con más variables que observaciones?}
  \vspace{0.2cm}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.30\textwidth}
      \centering
      {\color{upmblue}\fontsize{40}{44}\selectfont\bfseries $p\gg n$}\\[0.3cm]
      {\small más variables ($p$) que observaciones ($n$)}
    \end{column}
    \begin{column}{0.66\textwidth}
      \begin{block}{Kalisch y Bühlmann (2007)}
        El PC es consistente en alta dimensión si:
        \begin{itemize}
          \item el grafo es poco denso (pocos vecinos);
          \item los datos son gaussianos;
          \item el nivel de significación $\alpha$ disminuye con $n$.
        \end{itemize}
      \end{block}
      \begin{alertblock}{La letra pequeña: fidelidad fuerte \textperiodcentered{} Uhler et al.\ (2013)}
        La prueba exige que las dependencias reales no sean demasiado débiles.
        Las distribuciones que incumplen esta condición no son casos raros:
        pueden ser una fracción sorprendentemente grande. Por eso el PC es
        frágil con muestras finitas.
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

% ---------------------------------------------------------------------------
\begin{frame}[t]{(ii) PC-stable y dependencia del orden}
  \framesubtitle{Reordenar las columnas no debería cambiar el resultado.}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{El problema: PC original}
        \begin{itemize}
          \item Elimina una arista y actualiza los vecinos al instante.
          \item Las pruebas siguientes dependen de lo que ya se eliminó.
          \item Distinto orden de variables $\Rightarrow$ grafo distinto.
        \end{itemize}
      \end{alertblock}
    \end{column}
    \begin{column}{0.49\textwidth}
      \begin{block}{La solución: PC-stable \textperiodcentered{} Colombo y Maathuis (2014)}
        \begin{itemize}
          \item Congela la lista de vecinos al inicio de cada nivel.
          \item Todas las pruebas del nivel usan la misma lista.
          \item Esqueleto independiente del orden.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
  \vspace{0.2cm}
  \begin{exampleblock}{Para las flechas: CPC (Ramsey et al., 2006)}
    Orienta un colisor solo si todos los separadores coinciden; si no, lo
    marca como ambiguo.
  \end{exampleblock}
  \vfill
  {\small\color{upmorange}\textbf{Precio:} más pruebas
  $\;\longrightarrow\;$ (iv)}
\end{frame}

% ---------------------------------------------------------------------------
\begin{frame}[t]{(iii) Conjuntos de condicionamiento grandes}
  \framesubtitle{Condicionar en muchas variables deja pocos datos por celda.}
  \begin{columns}[T,onlytextwidth]

%  2.2  INTRODUCCIÓN
% ===========================================================================

\begin{frame}[t]{Pruebas de IC: el motor estadístico}
  \small
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.53\textwidth}
      \begin{itemize}
        \item Cada decisión de poda u orientación del PC se apoya en
              contrastes $H_0: I(X,Y\mid\mathbf Z)$.
        \item Bajo Markov causal y fidelidad:
      \end{itemize}
      \[
        I(X,Y\mid\mathbf Z)
        \;\Longleftrightarrow\;
        X\perp_G Y\mid\mathbf Z.
      \]
      \begin{itemize}
        \item Frente a las funciones de puntuación, que evalúan estructuras
              completas, la decisión es \alert{local y trazable}: cada arista
              se justifica con uno o varios contrastes y sus $p$-valores.
        \item No hay contraste universal: van de los paramétricos clásicos a
              los no paramétricos, y cada uno impone un modelo de dependencia
              y paga esa restricción con potencia.
      \end{itemize}
    \end{column}
    \begin{column}{0.44\textwidth}
      \begin{block}{Por qué importa}
        La calidad del grafo aprendido queda acotada por la calidad de las
        decisiones estadísticas.
      \end{block}
      \begin{alertblock}{Un contraejemplo}
        Si $X\sim\mathcal N(0,1)$ e $Y=X^2$:
        \[
          \mathrm{corr}(X,Y)=0
          \quad\text{pero}\quad
          \neg I(X,Y).
        \]
        La correlación parcial no la detecta.
      \end{alertblock}
    \end{column}
  \end{columns}
  {\scriptsize\color{upmgray}Fuentes: Spirtes et al. (2000); Kitson et al.
  (2023); Shah y Peters (2020).}
\end{frame}

% ===========================================================================
%  2.2.1  CORRELACIÓN PARCIAL
% ===========================================================================
\subsection{Correlación parcial}

\begin{frame}[t]{Correlación parcial: el contraste gaussiano}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.55\textwidth}
      \begin{itemize}
        \item Datos continuos con normalidad conjunta
              \srccite{Lauritzen, 1996}.
        \item $I(X,Y\mid\mathbf Z)$ equivale a que la correlación parcial
              $\rho_{XY\cdot\mathbf Z}$ se anule.
        \item Se estima la covarianza muestral $S$ de $(X,Y,\mathbf Z)$ y se
              invierte: $\hat K=S^{-1}$.
      \end{itemize}
      \[
        \hat\rho_{XY\cdot\mathbf Z}
        =\frac{-\hat K_{XY}}{\sqrt{\hat K_{XX}\hat K_{YY}}}.
      \]
      \begin{itemize}
        \item La transformada $z$ de Fisher convierte la correlación en un
              estadístico aproximadamente normal \srccite{Hotelling, 1953;
              Anderson, 2003}.
      \end{itemize}
    \end{column}
    \begin{column}{0.42\textwidth}
      \begin{block}{Contraste}
        \[
          z(r)=\tfrac12\ln\frac{1+r}{1-r},
        \]
        \[
          T=\sqrt{n-|\mathbf Z|-3}\;z(\hat\rho).
        \]
        Se rechaza $H_0$ si $|T|>z_{1-\alpha/2}$.
      \end{block}
      \begin{exampleblock}{Coste}
        $\mathcal O(|\mathbf Z|^3)$ por contraste, con $p$-valores
        analíticos.
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[t]{Correlación parcial: ventajas y limitaciones}
  \small
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      \begin{block}{Ventajas}
        \begin{itemize}
          \item Coste bajo: invertir el bloque de covarianzas.
          \item $p$-valores analíticos, sin remuestreo.
          \item Implementación de referencia en \texttt{pcalg} y
                \texttt{bnlearn} \srccite{Kalisch et al., 2012; Scutari, 2010}.
          \item Sostiene la consistencia del PC en alta dimensión con grafos
                dispersos \srccite{Kalisch y Bühlmann, 2007}.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.49\textwidth}
      \begin{alertblock}{Limitaciones}
        \begin{itemize}
          \item Solo es válido bajo normalidad y linealidad.
          \item Fuera de ese modelo, las versiones de rango solo aproximan
                \srccite{Baba et al., 2004}.
          \item Correlación nula no implica independencia en general.
          \item En muestras pequeñas, la aproximación asintótica del
                $p$-valor puede ser pobre \srccite{Tsamardinos et al., 2010}.
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}
  \vspace{0.2cm}
  \begin{exampleblock}{Líneas recientes}
    WLS para varianza no constante \srccite{Günther et al., 2022};
    correlaciones de rango y cópulas para no normalidad
    \srccite{Harris y Drton, 2012; Cui et al., 2016}; permutaciones para
    calibrar en muestra pequeña.
  \end{exampleblock}
\end{frame}

% ===========================================================================
%  2.2.2  TABLAS DE CONTINGENCIA
% ===========================================================================
\subsection{Tablas de contingencia}

\begin{frame}[t]{Tablas de contingencia: el contraste discreto}
  \small
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.57\textwidth}
      \begin{itemize}
        \item Datos discretos o categóricos: se comparan los conteos observados
              con los esperados bajo independencia
              \srccite{Spirtes et al., 2000; Kitson et al., 2023}.
        \item $n_{ijk}$: observaciones con $X=i$, $Y=j$ y $\mathbf Z=k$.
      \end{itemize}
      \begin{block}{Estadísticos}
        \[
          E_{ijk}=\frac{n_{i\cdot k}\,n_{\cdot jk}}{n_{\cdot\cdot k}},
        \]
        \[
          \chi^2=\sum_{i,j,k}\frac{(n_{ijk}-E_{ijk})^2}{E_{ijk}},
        \]
        \[
          G^2=2\sum_{i,j,k} n_{ijk}\ln\frac{n_{ijk}}{E_{ijk}}.
        \]
      \end{block}
    \end{column}
    \begin{column}{0.40\textwidth}
      \begin{block}{Distribución nula}
        \[
          \nu=(I-1)(J-1)\prod_l K_l
        \]
        Se rechaza $H_0$ si el estadístico supera
        $\chi^2_{\nu,1-\alpha}$ \srccite{Spirtes et al., 2000}.
      \end{block}
      \begin{exampleblock}{Lectura}
        $G^2=2n\,\hat I(X,Y\mid\mathbf Z)$: versión discreta del contraste de
        razón de verosimilitudes, dentro de la familia de información mutua
        \srccite{Scutari, 2010}.
      \end{exampleblock}
      {\scriptsize\color{upmgray}Ordinales: tendencia de Jonckheere--Terpstra
      \srccite{Jonckheere, 1954; Terpstra, 1952}.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[t]{Tablas de contingencia: ventajas y limitaciones}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      \begin{block}{Ventajas}
        \begin{itemize}
          \item Contraste canónico para datos categóricos
                \srccite{Kitson et al., 2023}.
          \item $p$-valores analíticos.
          \item Lectura directa como divergencia entre la tabla observada y la
                esperada.
          \item En las simulaciones originales del PC, $G^2$ acertó el grafo
                más veces que $\chi^2$ \srccite{Spirtes et al., 2000}.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.49\textwidth}
      \begin{alertblock}{Limitaciones}
        \begin{itemize}
          \item Celdas $I\cdot J\cdot\prod_l K_l$: crecimiento exponencial con
                $|\mathbf Z|$ y la cardinalidad.
          \item Con pocas observaciones por celda, la aproximación $\chi^2$
                deja de ser fiable.
          \item Regla práctica: $n\ge 5\times$ número de celdas
                \srccite{Spirtes et al., 2000}.
          \item El coste es lineal en celdas, pero las celdas son el cuello de
                botella.
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}
  \vspace{0.1cm}
  \begin{exampleblock}{Líneas recientes}
    Tablas suavizadas (Dash et al., 2003); SECMI (Kubkowski et al., 2021);
    permutaciones.
  \end{exampleblock}
\end{frame}

% ===========================================================================
\subsection{Regresión y métodos de residuos}

\begin{frame}[t]{Regresión y residuos: el contraste GCM}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.53\textwidth}
      \begin{itemize}
        \item Datos mixtos: no se modela directamente la dependencia, sino que
              se elimina el efecto de $\mathbf Z$.
        \item La versión general es la \alert{medida de covarianza
              generalizada (GCM)} de Shah \& Peters (2020).
        \item Se ajustan $\hat m_X(\mathbf z)$ y $\hat m_Y(\mathbf z)$ y se
              toman los residuos
              $r_i=x_i-\hat m_X(\mathbf z_i)$ y
              $s_i=y_i-\hat m_Y(\mathbf z_i)$.
        \item Bajo independencia, los productos $R_i=r_i s_i$ tienen media
              cero.
      \end{itemize}
      \[
        T_n=\frac{\sqrt n\,\bar R}{\hat\sigma_R},
        \qquad
        \hat\sigma_R^2=\frac1n\sum_{i=1}^n(R_i-\bar R)^2.
      \]
    \end{column}
    \begin{column}{0.44\textwidth}
      \begin{block}{Distribución nula}
        $T_n\longrightarrow\mathcal N(0,1)$ si el producto de los errores
        cuadráticos medios de ambas regresiones es $o(n^{-1})$
        \srccite{Shah y Peters, 2020}.
      \end{block}
      \begin{exampleblock}{Caso lineal-gaussiano}
        El contraste coincide, salvo normalización, con la correlación parcial
        de Fisher \srccite{Shah y Peters, 2020}.
      \end{exampleblock}
      \begin{alertblock}{Idea}
        No fija la forma de la dependencia, sino la de las medias
        condicionales.
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[t]{Regresión y residuos: ventajas y limitaciones}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      \begin{block}{Ventajas}
        \begin{itemize}
          \item Flexible: lineal, GLM, bosques y kernels; admite datos mixtos.
          \item Cubre parte de la no linealidad.
          \item WGCM pondera por la varianza \srccite{Scheidegger et al., 2022};
                el doble ML evita el sobreajuste \srccite{Chernozhukov et al.,
                2018}.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.49\textwidth}
      \begin{alertblock}{Limitaciones}
        \begin{itemize}
          \item Las regresiones no son neutrales: solo detectan alternativas
                con covarianza residual.
          \item Sin restringir la clase nula, ningún contraste con nivel
                uniforme tiene potencia (Shah \& Peters, 2020).
          \item Elegir las regresiones exige conocer el problema; su coste
                puede ser alto.
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}
  \begin{exampleblock}{Líneas recientes}
    WGCM amplía las alternativas a costa de potencia
    \srccite{Scheidegger et al., 2022}; el doble ML permite regresiones
    flexibles y calibradas \srccite{Chernozhukov et al., 2018}.
  \end{exampleblock}
\end{frame}

%  2.2.4  MÉTODOS NO PARAMÉTRICOS
% ===========================================================================
\subsection{Métodos no paramétricos}

\begin{frame}[t]{Kernel y vecinos: contrastes no paramétricos}
  \small
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      \begin{block}{Kernel: HSIC y KCI}
        \begin{itemize}
          \item HSIC: dos variables son independientes si y solo si ninguna
                función de una covaría con ninguna función de la otra
                \srccite{Gretton et al., 2005}.
          \item KCI elimina $\mathbf Z$ por regresión no paramétrica y aplica
                una medida tipo HSIC a los residuos \srccite{Zhang et al., 2011}.
          \item Nulo por Monte Carlo o aproximación gamma
                \srccite{Zhang et al., 2011}.
        \end{itemize}
        \[
          \mathrm{HSIC}(X,Y)=0\iff X\perp Y.
        \]
      \end{block}
    \end{column}
    \begin{column}{0.49\textwidth}
      \begin{block}{Vecinos más cercanos}
        \begin{itemize}
          \item La información mutua condicional se estima con un estimador
                de $k$ vecinos \srccite{Runge, 2018}.
          \item El contraste se calibra con permutaciones locales
                \srccite{Runge, 2018}.
          \item RCIT/RCoT sustituyen los kernels por características de
                Fourier aleatorias y pasan a coste lineal
                \srccite{Strobl et al., 2019}.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
  \vspace{-2pt}
  {\small
  \begin{alertblock}{Idea común}
    No se fija la forma de la dependencia ni la de las medias condicionales.
  \end{alertblock}}
\end{frame}

\begin{frame}[t]{Métodos no paramétricos: ventajas y limitaciones}
  \small
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      \begin{block}{Ventajas}
        \begin{itemize}
          \item No asumen forma funcional.
          \item Captan dependencias no lineales.
          \item Con densidades no suaves, los vecinos superan en potencia a
                los métodos de kernel \srccite{Runge, 2018}.
          \item Los de kernel mitigan la maldición de la dimensionalidad
                \srccite{Zhang et al., 2011}; RCIT/RCoT escalan a muestras
                grandes \srccite{Strobl et al., 2019}.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.49\textwidth}
      \begin{alertblock}{Limitaciones}
        \begin{itemize}
          \item KCI es $\mathcal O(n^2)$ o peor: inviable en muestras grandes
                \srccite{Strobl et al., 2019}.
          \item La distribución nula analítica no está bien calibrada por
                debajo de $\sim 1000$ muestras \srccite{Runge, 2018}.
          \item Las permutaciones mejoran la calibración a costa de potencia
                \srccite{Runge, 2018}.
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}
  \vspace{-0.1cm}
  \begin{exampleblock}{Líneas recientes}
    Contrastes \emph{distribution-free} cuyo índice se anula si y solo si hay
    independencia \srccite{Cai et al., 2022}; permutaciones aprendidas para
    estimar mejor el nulo \srccite{Doran et al., 2014}.
  \end{exampleblock}
\end{frame}

% ===========================================================================
%  2.2.5  EXTENSIONES PARA CAUSALIDAD
% ===========================================================================
\subsection{Extensiones para causalidad}

\begin{frame}[t]{De la independencia a la causalidad}
  \small
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.47\textwidth}
      \begin{block}{El problema de encadenar contrastes}
        \begin{itemize}
          \item Ni $\alpha$ ni la potencia miden el error de la búsqueda, que
                depende del número de contrastes y de la estructura recorrida
                \srccite{Spirtes et al., 2000}.
          \item Una eliminación errónea inhibe contrastes posteriores y
                arrastra esqueleto y orientación
                \srccite{Colombo y Maathuis, 2014}.
          \item La consistencia exige $\alpha_n\to 0$ con $n$
                \srccite{Kalisch y Bühlmann, 2007}.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.50\textwidth}
      \begin{block}{Nuevos criterios de separación}
        \begin{itemize}
          \item \textbf{$\sigma$-separación}: mecanismos no lineales y ciclos
                \srccite{Forré y Mooij, 2018}.
          \item \textbf{$m$-separación} en MAG/PAG: variables latentes y sesgo
                de selección (FCI) \srccite{Richardson y Spirtes, 2002}.
        \end{itemize}
      \end{block}
      \begin{block}{Nuevos escenarios}
        \begin{itemize}
          \item \textbf{COmbINE}: unifica datos observacionales e
                intervencionales parcialmente solapados
                \srccite{Triantafillou y Tsamardinos, 2015}.
          \item \textbf{CTBN}: contrastes para trayectorias continuas; explotan
                la asimetría del tiempo \srccite{Villa y Stella, 2016}.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[t]{Extensiones causales: alcance y límites}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      \begin{block}{Ventajas}
        \begin{itemize}
          \item Dotan a los contrastes de semántica causal plena.
          \item Cubren ciclos, latentes y sesgo de selección.
          \item Integran datos heterogéneos y temporales.
          \item FCI/RFCI y COmbINE amplían el marco sin abandonar las

          \item Un solo error estadístico altera los conjuntos de

      \end{alertblock}
    \end{column}
  \end{columns}

\section[Enfoques híbridos]{Enfoques híbridos y líneas futuras}
\subsection{Motivación y arquitectura}

% 1 / 00:55. Idea oral: los enfoques tradicionales fallan por razones distintas.
\begin{frame}[t]{Enfoques híbridos: motivación}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      \begin{block}{Métodos \textit{constraint-based}}
        \begin{itemize}
          \item Construyen relaciones a partir de contrastes de independencia condicional (IC).
          \item Coste computacional crece con el tamaño y la densidad del grafo.
          \item Una decisión estadística errónea puede propagarse durante la búsqueda.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.49\textwidth}
      \begin{block}{Métodos \textit{score-and-search}}
        \begin{itemize}
          \item Evalúan estructuras mediante funciones de puntuación, como BIC.
          \item El espacio de DAGs crece de forma superexponencial con las variables.
          \item Las búsquedas heurísticas pueden terminar en óptimos locales.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
  \vspace{0.16cm}
  \begin{exampleblock}{Motivación de los enfoques híbridos}
    Usar información estadística sobre las relaciones entre variables para \alert{orientar o restringir} una búsqueda basada en puntuaciones.
  \end{exampleblock}
  {\scriptsize\color{upmgray}Kitson et al. (2023).}
\end{frame}

\subsection{Evolución de Restrict/Maximise}

% 3 / 01:00. CB es iterativo; PC aproxima restricciones y K2 necesita orden.
\begin{frame}[t]{Constraint-Bayesian: PC + K2}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.54\textwidth}
      \begin{itemize}
        \item Uno de los primeros algoritmos híbridos de aprendizaje estructural.
        \item Utiliza PC para obtener información sobre dependencias y una ordenación de las variables.
        \item K2 busca un DAG mediante puntuaciones, respetando esa ordenación.
        \item \alert{Itera ambas fases}: incrementa el tamaño permitido de los conjuntos condicionantes mientras mejora el resultado.
      \end{itemize}
    \end{column}

  \begin{block}{Procedimiento}

      \node[paso,below=of orden] (k2) {\textbf{K2}: búsqueda voraz};

% 4 / 01:00. Cambio clave: no se condiciona todo a un orden global, sino a

\begin{frame}[t]{Sparse Candidate: selección de padres}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.54\textwidth}
      \begin{itemize}
        \item Para cada variable $X_i$, selecciona un conjunto reducido de posibles padres $C_i$.
        \item Los candidatos se eligen mediante criterios estadísticos de relevancia y dependencia.
        \item Realiza una búsqueda por puntuación restringida \textit{Hill-Climbing} con lista tabú.
        \item \alert{Alterna} selección de candidatos y búsqueda, actualizando los conjuntos con el DAG obtenido.
      \end{itemize}
    \end{column}
    \begin{column}{0.43\textwidth}
  \begin{block}{Procedimiento iterativo}
    \centering

      node distance=0.12cm,

\end{frame}

\begin{frame}[t]{MMHC: aprendizaje en dos etapas}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.54\textwidth}
      \begin{itemize}
        \item \textbf{MMPC:} pruebas de IC para identificar
        posibles padres e hijos de cada variable.

        \item Los conjuntos locales estiman las adyacencias
        del \textbf{esqueleto} de la red.

        \item \textbf{Hill-Climbing:} construye un DAG
        mediante puntuaciones, restringiendo la incorporación
        de aristas a las relaciones candidatas.
      \end{itemize}

      \vspace{0.47cm}

      \begin{exampleblock}{Aportación frente a Sparse Candidate}
        \small
        Mejora la selección de candidatos mediante

      \end{exampleblock}

    \end{column}

    \begin{column}{0.43\textwidth}
      \begin{block}{Dos fases diferenciadas}
        \centering

        \textbf{MMPC}\\
        {\small Identificación de posibles padres e hijos}\\[0.12cm]

        $\Downarrow$\\[0.12cm]

        \textbf{Hill-Climbing}\\
        {\small Búsqueda por puntuaciones restringida
        a las relaciones candidatas}
      \end{block}

      \begin{alertblock}{Dependencia del filtro}
        \small
        Si MMPC omite una relación verdadera,
        Hill-Climbing no puede recuperarla.
      \end{alertblock}
    \end{column}
  \end{columns}

\end{frame}

\subsection{Mejoras en las restricciones}

% 7 / 01:00. Relacion falsos negativos -> errores irrecuperables. H2PC propone

\begin{frame}[t]{H2PC: mejorar la selección de candidatos}
  \begin{columns}[T,onlytextwidth]

    \begin{column}{0.53\textwidth}
      \begin{itemize}

        \item Sustituye \textbf{MMPC} por
        \textit{Hybrid Parents and Children}
        (\textbf{HPC}), que combina
        procedimientos de aprendizaje local.

        \item Busca reducir los
        \alert{falsos negativos} durante la
        selección de candidatos, evitando
        descartar relaciones verdaderas.

        \item Los experimentos originales
        mostraron mejoras en la
        \textbf{calidad estructural}
        frente a MMHC.
      \end{itemize}
    \end{column}

    \begin{column}{0.44\textwidth}

      \begin{block}{Aportación frente a MMHC}
        Mejora la identificación de relaciones
        candidatas sin modificar la arquitectura
        general de aprendizaje.
      \end{block}

      \begin{alertblock}{Limitación}
        La selección más exhaustiva de candidatos
        incrementa el coste computacional,
        obteniendo mayores tiempos de ejecución
        que MMHC en los experimentos originales.
      \end{alertblock}

    \end{column}
  \end{columns}

  {\scriptsize\color{upmgray}Gasse et al. (2014).}
\end{frame}
% 8 / 01:05. ARGES mejora como se aplica la restriccion durante la busqueda.
\begin{frame}[t]{ARGES: restricciones adaptativas}
  \begin{columns}[T,onlytextwidth]

    \begin{column}{0.53\textwidth}
      \begin{itemize}
        \item Parte de \textit{Greedy Equivalence Search}
        (\textbf{GES}), que busca sobre clases de
        equivalencia de Markov.

        \item Restringe inicialmente la búsqueda
        utilizando un esqueleto o un grafo de
        independencias condicionales estimado.

        \item Permite añadir aristas inicialmente excluidas cuando aparecen determinadas configuraciones en el grafo.
        \item Bajo ciertos supuestos, presenta
        \textbf{garantías de consistencia},
        incluso en escenarios de alta dimensión.
      \end{itemize}
    \end{column}

    \begin{column}{0.44\textwidth}

      \begin{block}{Aportación frente a H2PC}
        En lugar de mejorar únicamente las
        restricciones iniciales, ARGES
        \textbf{flexibiliza su aplicación}
        durante la optimización.
      \end{block}

      \begin{alertblock}{Limitación}
        La relajación solo permite determinadas
        aristas adicionales y no garantiza
        encontrar la estructura óptima
        con muestras finitas.
      \end{alertblock}

    \end{column}
  \end{columns}

  {\scriptsize\color{upmgray}Nandy et al. (2018).}
\end{frame}

% 9 / 00:50. Unir los dos articulos en la estructura ventaja-limitacion que

\begin{frame}[t]{Restricciones: ventajas y limitaciones}
  \begin{columns}[T,onlytextwidth]

    \begin{column}{0.48\textwidth}
      \begin{block}{Mejor selección: H2PC}
        \begin{itemize}
          \item Detecta más relaciones candidatas mediante HPC.
          \item Reduce el riesgo de perder aristas en el primer paso.
          \item A cambio, incrementa el trabajo estadístico previo.
        \end{itemize}
      \end{block}
    \end{column}

    \begin{column}{0.49\textwidth}
      \begin{block}{Mayor flexibilidad: ARGES}
        \begin{itemize}
          \item Mantiene una búsqueda inicialmente restringida.
          \item Admite ciertas incorporaciones adaptativas de aristas.
          \item Sus garantías dependen de supuestos específicos.
        \end{itemize}
      \end{block}
    \end{column}

  \end{columns}

  \vspace{0.2cm}

  \begin{exampleblock}{Más allá de Restrict/Maximise}
    Existen otras estrategias híbridas que combinan
    restricciones y puntuaciones sin seguir necesariamente
    este orden de aprendizaje.
  \end{exampleblock}

\subsection{Estrategias alternativas}

% 10 / 01:05. Diferenciar CPDAG de PAG: el segundo representa equivalencias
% cuando se permiten posibles variables latentes y seleccion.
\begin{frame}[t]{GFCI: descubrimiento causal con variables latentes}
  \begin{columns}[T,onlytextwidth]

    \begin{column}{0.53\textwidth}
      \begin{itemize}

        \item A diferencia de \textit{Restrict/Maximise},
        comienza con una búsqueda por puntuaciones
        y después aplica restricciones.

        \item Ejecuta \textit{Fast Greedy Search}
        (\textbf{FGS}), obteniendo un
        \textbf{CPDAG} inicial.

        \item Utiliza pruebas de IC y reglas de
        \textbf{FCI} para refinar las adyacencias
        y orientaciones obtenidas.

        \item Produce un \alert{PAG}, que representa
        relaciones causales compatibles con la
        existencia de posibles variables latentes.

      \end{itemize}
    \end{column}

    \begin{column}{0.44\textwidth}

  \begin{block}{Procedimiento}
    \centering
    \small

    \textbf{FGS (Score-based)}\\
    {\scriptsize CPDAG inicial}\\[0.05cm]

    $\Downarrow$\\[0.05cm]

    \textbf{Refinamiento tipo FCI}\\
    {\scriptsize Tests de IC y orientación}\\[0.05cm]

    $\Downarrow$\\[0.05cm]

    \textbf{PAG final}

  \end{block}

  \vspace{-0.12cm}

  \begin{exampleblock}{CPDAG vs. PAG}
    \small
    \centering

    \textbf{CPDAG:} $A-B$\\[-0.04cm]
    {\scriptsize Orientación no determinada.}\\[0.07cm]

    \textbf{PAG:} $A\leftrightarrow B$\\[-0.04cm]
    {\scriptsize Compatible con confusión latente.}

  \end{exampleblock}

\end{column}
  \end{columns}

  {\scriptsize\color{upmgray}
  Ogarrio et al. (2016). PAG: Partial Ancestral Graph.}
\end{frame}

% 11 / 01:05. Supuesto de conectividad es útil solo si dominio lo justifica.
\begin{frame}[t]{SaiyanH: incorporar conectividad}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.53\textwidth}
      \begin{itemize}
        \item Diseñado para dominios donde se espera una red
        \textbf{conectada}.

        \item Construye inicialmente un grafo no dirigido y
        conectado utilizando medidas de asociación.

        \item Orienta relaciones con pruebas de IC,
        puntuaciones y criterios asociados a intervenciones.

        \item Refina el resultado mediante \textit{Hill-Climbing}
        con memoria tabú y restricciones de conectividad.
      \end{itemize}
    \end{column}

    \begin{column}{0.44\textwidth}
      \begin{block}{Aportación}
        Incorpora conocimiento estructural del dominio
        que otros enfoques híbridos no exigen.
      \end{block}

      \begin{alertblock}{Riesgo}
        Si la red verdadera tiene componentes desconectadas,
        forzar conectividad puede introducir aristas espurias.
      \end{alertblock}
    \end{column}
  \end{columns}

  \vfill

  {\scriptsize\color{upmgray}Constantinou (2020).}
  \vspace{0.00cm}

\end{frame}

\subsection{Avances y desafíos abiertos}

  \begin{columns}[T,onlytextwidth]

      \begin{block}{ESLH}
        \begin{itemize}
          \item Umbrales dinámicos (DTH) para reducir pruebas poco prometedoras.

  \end{alertblock}

% 13 / 01:00. No atribuir superioridad universal a ninguna familia.
\begin{frame}[t]{¿Son mejores los métodos híbridos?}
  \begin{columns}[T,onlytextwidth]

    \begin{column}{0.54\textwidth}
      \begin{itemize}
        \item Una evaluación de 2026 comparó
        \textbf{14 algoritmos} \textit{constraint-based},
        \textit{score-based} e híbridos.

        \item Utilizó \textbf{cinco DAGs simulados},
        diferentes tamaños muestrales y
        métricas estructurales.

        \item Consideró \textbf{MCC} como métrica principal,
        junto con \textbf{F1} y \textbf{SHD}

        \item Ninguna familia de algoritmos
        \alert{dominó sistemáticamente}
        en todos los escenarios.
      \end{itemize}
    \end{column}

    \begin{column}{0.43\textwidth}
      \begin{block}{Observación}
        MMHC y H2PC obtuvieron peores resultados

        el tamaño muestral y la métrica utilizada.
      \end{alertblock}
    \end{column}

\begin{frame}[t]{Conclusiones}
  \footnotesize
  \begin{itemize}\setlength{\itemsep}{2pt}
    \item \textbf{No existe una estrategia universalmente superior.} El
          rendimiento depende de la interacción entre potencia estadística,
          coste computacional y supuestos causales.
    \item \textbf{El PC es auditable, pero frágil.} Cada decisión es trazable;
          en muestra finita, un error se propaga al esqueleto y a la
          orientación. Las limitaciones se desplazan, no se eliminan.
    \item \textbf{Las pruebas de IC acotan lo que puede aprenderse.} Ningún
          contraste tiene potencia uniforme frente a todas las alternativas sin
          restringir la clase nula: no hay una prueba universal.
    \item \textbf{La independencia no basta para orientar.} Bajo Markov,
          fidelidad y suficiencia, el PC devuelve como máximo un CPDAG; orientar
          más exige supuestos, intervenciones o conocimiento de dominio.
    \item \textbf{Los híbridos no garantizan una mejora.} Un falso negativo en
          el filtro deja aristas irrecuperables; la clave está en diseñar la
          interfaz entre restricciones y puntuaciones.
  \end{itemize}
  \vfill
  \begin{exampleblock}{La pregunta relevante}
    \centering No qué algoritmo es mejor en abstracto, sino \alert{qué supuestos
    son defendibles} y \alert{qué errores pueden tolerarse} en cada dominio.
  \end{exampleblock}
  {\scriptsize\color{upmgray}Revisión narrativa: las cifras proceden de los
  estudios originales, no de una reproducción independiente.}
\end{frame}

\begin{frame}[t]{Líneas futuras}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      \begin{block}{1. Garantías en muestra finita}
        \small Cotas, conjuntos de confianza sobre el CPDAG y calibración
        adaptativa del nivel de significación.
      \end{block}
      \begin{block}{2. Pruebas adaptativas}
        \small Seleccionar o combinar contrastes según el dato, $n$ y
        $|\mathbf S|$; extensión a datos mixtos y de alta dimensión.
      \end{block}
      \begin{block}{3. Supuestos relajados}
        \small Variables latentes, ciclos, no estacionariedad y eventos
        extremos, con garantías formales y datos intervencionales.
      \end{block}
    \end{column}
    \begin{column}{0.49\textwidth}
      \begin{block}{4. Escalabilidad y enfoques diferenciables}
        \small Paralelización, algoritmos \emph{anytime} y relajaciones
        diferenciables que conserven garantías discretas.
      \end{block}
      \begin{block}{5. Evaluación reproducible}
        \small Bancos de prueba comunes que varíen topología, muestra,
        dimensionalidad y confusión latente, midiendo calidad estructural,
        calibración y coste.
      \end{block}
      \begin{exampleblock}{Cierre}
        \small Las garantías condicionan el diseño de las pruebas; relajar
        supuestos exige nuevas formas de evaluar la identificabilidad.
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}
